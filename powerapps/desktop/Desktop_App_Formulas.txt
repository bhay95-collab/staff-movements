// =====================================================================
// App > Formulas   (desktop app)  paste the whole thing
// Everything on the screens is drawn for a 1366 x 768 window and then stretched with these.
// =====================================================================

// across and down
SX = App.Width / 1366;
SY = App.Height / 768;

// text: follows the smaller of the two, never below 75% or above 140%
SF = Max(0.75, Min(1.4, Min(SX, SY)));

// corner radii and gaps: follow the smaller of the two so rounded buttons stay round
SR = Min(SX, SY);

// ---- who is using the app (used by the Training screens)
MeEmail = Lower(User().Email);
MeName = User().FullName;

// ---- Training: who can add dates, remove people and email attendees (set in the TrainingAppPermissions list)
CanManageTraining = Coalesce(LookUp(TrainingAppPermissions, Lower(Person.Email) = MeEmail).Access.Value, "User") in ["Director", "Team Leader", "HP4"];
