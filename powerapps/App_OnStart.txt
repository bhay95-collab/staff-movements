// Old equipment warning: equipment signed out for over 6 weeks
Set(
    colOldAllocations,
    Filter(
        Equipment_Inventory_SharePoint,
        Status.Value = "Allocated",
        Physiotherapist = User().FullName,
        DateDiff('Date Signed Out', Today()) > 42
    )
);

If(
    CountRows(colOldAllocations) > 0,
    Notify(
        "You have equipment signed out for over 6 weeks. Please review:" &
        Concat(colOldAllocations, Char(10) & 'Patient Name' & " - " & Title),

    )
);
