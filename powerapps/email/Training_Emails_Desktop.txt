// Training emails (Desktop, screen Training_Sessions). Needs EmailHead and EmailFoot (App > Formulas).
// Paste each formula into the OnSelect of the named (hidden) button on the Training_Sessions screen.

// ===== btnDoSend_Trs   (Email attendees > Send email)
With(
    { to: Concat(Filter(colTrBookings, SessionID = trSel.SesKey), BookedUserEmail, ";") },
    If(
        IsBlank(to),
        Notify("Nobody is booked into this date yet.", NotificationType.Warning),
        IfError(
            With(
                {
                    subj: Trim(txtEmailSubject_Trs.Text),
                    msg: Substitute(Substitute(Substitute(Substitute(txtEmailBody_Trs.Text, "&", "&amp;"), "<", "&lt;"), Char(13), ""), Char(10), "<br>")
                },
                Office365Outlook.SendEmailV2(
                    to,
                    subj,
                    Substitute(Substitute(EmailHead, "{{TITLE}}", Substitute(Substitute(subj, "&", "&amp;"), "<", "&lt;")), "{{SUB}}", trType & "&nbsp;&nbsp;|&nbsp;&nbsp;" & Text(trSel.SessionDate, "ddd d mmm yyyy") & "&nbsp;&nbsp;|&nbsp;&nbsp;" & Text(trSel.SessionDate, "h:mm AM/PM")) &
                    "<tr><td style='padding:28px 32px 0 32px;font-size:15px;line-height:23px;color:#3A5068;'>" & msg & "</td></tr>" &
                    "<tr><td style='padding:20px 32px 0 32px;'><div style='font-size:11px;font-weight:700;letter-spacing:1px;color:#8C9BAE;padding:0 0 8px 2px;'>SESSION</div><table role='presentation' width='100%' cellpadding='0' cellspacing='0' border='0' style='border-collapse:separate;border-spacing:0;width:100%;border:1px solid #E3EAF1;border-radius:14px;overflow:hidden;'>" & "<tr><td style='width:190px;padding:10px 14px;background-color:#F7FAFC;font-size:11px;font-weight:600;letter-spacing:1px;color:#8C9BAE;'>COURSE</td><td style='padding:10px 14px;font-size:14px;color:#0E1C2A;'>" & trType & "</td></tr>" & "<tr><td style='width:190px;padding:10px 14px;background-color:#F7FAFC;font-size:11px;font-weight:600;letter-spacing:1px;color:#8C9BAE;border-top:1px solid #E3EAF1;'>DATE</td><td style='padding:10px 14px;font-size:14px;color:#0E1C2A;border-top:1px solid #E3EAF1;'>" & Text(trSel.SessionDate, "dddd d mmmm yyyy") & "</td></tr>" & "<tr><td style='width:190px;padding:10px 14px;background-color:#F7FAFC;font-size:11px;font-weight:600;letter-spacing:1px;color:#8C9BAE;border-top:1px solid #E3EAF1;'>TIME</td><td style='padding:10px 14px;font-size:14px;color:#0E1C2A;border-top:1px solid #E3EAF1;'>" & Text(trSel.SessionDate, "h:mm AM/PM") & "</td></tr>" & "<tr><td style='width:190px;padding:10px 14px;background-color:#F7FAFC;font-size:11px;font-weight:600;letter-spacing:1px;color:#8C9BAE;border-top:1px solid #E3EAF1;'>LOCATION</td><td style='padding:10px 14px;font-size:14px;color:#0E1C2A;border-top:1px solid #E3EAF1;'>" & Coalesce(trSel.Location, "To be confirmed") & "</td></tr>" & "</table></td></tr>" &
                    "<tr><td style='padding:8px 32px 20px 32px;'>&nbsp;</td></tr>" &
                    Substitute(EmailFoot, "{{NOTE}}", "You are receiving this because you are booked into this session."),
                    { IsHtml: true }
                )
            );
            Set(trShowEmail, false);
            Notify("Email sent to the booked attendees.", NotificationType.Success),
            Notify("The email could not be sent. Please try again.", NotificationType.Error)
        )
    )
)
