    ForAll(
        ParseJSON(varTeamBriefJson) As fileRow,
        Collect(
            colTeamBriefFiles,
            {
                Id:        Text(fileRow.ItemInternalId),
                Name:      Text(fileRow.'{Name}'),
                FileName:  Text(fileRow.'{FilenameWithExtension}'),
                // Folder path only, e.g. "Shared Documents/Team Brief/" or ".../Team Brief/Archive/"
                Path:      Text(fileRow.'{Path}'),
                // Full path including file/folder name, e.g. ".../Team Brief/Archive/" or ".../file.pdf"
                FullPath:  Text(fileRow.'{FullPath}'),
                Link:      Text(fileRow.'{Link}'),
                // True for folders, false for files
                IsFolder:  fileRow.'{IsFolder}'
            }
        )
    )
);
